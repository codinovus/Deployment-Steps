Deployment Steps

1. Update Upgrade

sudo apt update
sudo apt upgrade -y

2. Install NVM

curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.3/install.sh | bash

source ~/.bashrc

3. Install Node

nvm install --lts

4. Install Postgreql

sudo apt install postgresql postgresql-contrib

sudo systemctl start postgresql
sudo systemctl enable PostgreSQL

5. Install NGNIX

sudo apt install nginx

sudo systemctl start nginx
sudo systemctl enable nginx

6. Create Configuration for Domain (For Frontend)

sudo nano /etc/nginx/sites-available/{ Your Domain Name }

server {
    listen 80;
    server_name { Your Domain Name } www.{ Your Domain Name };
    
    # Serve static files from browser folder
    location / {
        root /var/www/{ Your Domain Name }/browser;
        try_files $uri $uri/ @server;
    }
    
    # Forward to SSR server when static file not found
    location @server {
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}


7. Enable Configuration

sudo ln -s /etc/nginx/sites-available/{ Your Domain Name }/etc/nginx/sites-enabled/
sudo ln -s /etc/nginx/sites-available/{ Your Domain Name }/etc/nginx/sites-enabled/

8. Test NGNIX COnfiguration

sudo nginx -t

if success

sudo systemctl restart nginx

9. Build Your Angular Application

10. Paste the angular application in 

/var/www/{ Your Domain Name }/dist/

if folder dosenot exists create it 

start server 

cd /var/www/bot1001.ecomtgbot.tech
pm2 start server/server.mjs --name angular-ssr

11. Reload NGNIX

sudo systemctl reload nginx

12. Install SSL Provider

sudo apt update
sudo apt install certbot python3-certbot-nginx

13. Get and Save Certificate

sudo certbot --nginx -d { Your Domain Name } -d www.{ Your Domain Name }

14. Update NGNIX Configuration

server {
    listen 80;
    server_name { Your Domain Name };
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name { Your Domain Name };
    
    ssl_certificate /etc/letsencrypt/live/{ Your Domain Name }/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/{ Your Domain Name }/privkey.pem;
    
    # Rest of your configuration
    location / {
        root /var/www/{ Your Domain Name }/browser;
        try_files $uri $uri/ @server;
    }
    
    location @server {
        proxy_pass http://localhost:4000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}


15. Reload NGNIX

sudo nginx -t
sudo systemctl reload nginx
